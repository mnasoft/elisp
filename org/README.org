  #!/usr/bin/env bash

  set -e

  HUNSPELL_DIR="/c/msys64/ucrt64/share/hunspell"
  TMP_DIR="/tmp/hunspell-dicts"

  mkdir -p "$TMP_DIR"
  mkdir -p "$HUNSPELL_DIR"

#  echo "Downloading Ukrainian dictionary..."
#  curl -L -o "$TMP_DIR/uk_UA.oxt" \
#    https://extensions.libreoffice.org/assets/downloads/ukrainian-dictionary-pack.oxt

#  echo "Downloading Russian dictionary..."
#  curl -L -o "$TMP_DIR/ru_RU.oxt" \
#    https://extensions.libreoffice.org/assets/downloads/russian-dictionary-pack.oxt

  echo "Extracting Ukrainian dictionary..."
  unzip -o "$TMP_DIR/uk_UA.oxt" -d "$TMP_DIR/uk_UA"
  cp "$TMP_DIR/uk_UA"/*.aff "$HUNSPELL_DIR/uk_UA.aff"
  cp "$TMP_DIR/uk_UA"/*.dic "$HUNSPELL_DIR/uk_UA.dic"

  echo "Extracting Russian dictionary..."
  unzip -o "$TMP_DIR/ru_RU.oxt" -d "$TMP_DIR/ru_RU"
  cp "$TMP_DIR/ru_RU"/*.aff "$HUNSPELL_DIR/ru_RU.aff"
  cp "$TMP_DIR/ru_RU"/*.dic "$HUNSPELL_DIR/ru_RU.dic"

  echo "Done. Installed dictionaries:"
  ls -1 "$HUNSPELL_DIR"

  echo "Testing hunspell -D..."
  hunspell -D
#+end_src
